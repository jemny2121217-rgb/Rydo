using UnityEngine; // ========================================== // 1. MOBILE INPUT MANAGER // Handles touch screen drag for camera look and fire button states. // ========================================== public class MobileInputManager : MonoBehaviour { public static MobileInputManager Instance; [HideInInspector] public Vector2 moveInput; [HideInInspector] public Vector2 lookInput; [HideInInspector] public bool isFiring; private int lookTouchId = -1; void Awake() { Instance = this; } void Update() { for (int i = 0; i < Input.touchCount; i++) { Touch t = Input.GetTouch(i); if (t.phase == TouchPhase.Began && t.position.x > Screen.width / 2f && lookTouchId == -1) { lookTouchId = t.fingerId; } if (t.fingerId == lookTouchId) { if (t.phase == TouchPhase.Moved) { lookInput = t.deltaPosition; } else if (t.phase == TouchPhase.Ended || t.phase == TouchPhase.Canceled) { lookTouchId = -1; lookInput = Vector2.zero; } } } } public void SetFireState(bool state) { isFiring = state; } } // ========================================== // 2. PLAYER CONTROLLER & SHOOTING // Manages movement, camera controls, gravity, and raycast shooting. // ========================================== [RequireComponent(typeof(CharacterController))] public class PlayerController : MonoBehaviour { public CharacterController controller; public Camera playerCamera; [Header("Movement Settings")] public float speed = 6f; public float gravity = -9.81f; public float mouseSensitivity = 0.2f; [Header("Combat Settings")] public float fireRate = 0.15f; public float damage = 20f; public float range = 100f; private float xRotation = 0f; private Vector3 velocity; private float nextTimeToFire = 0f; void Update() { HandleCameraLook(); HandleMovement(); HandleShooting(); } void HandleCameraLook() { Vector2 look = MobileInputManager.Instance != null ? MobileInputManager.Instance.lookInput : Vector2.zero; if (look == Vector2.zero) { look = new Vector2(Input.GetAxis("Mouse X"), Input.GetAxis("Mouse Y")) * 10f; } float mouseX = look.x * mouseSensitivity; float mouseY = look.y * mouseSensitivity; xRotation -= mouseY; xRotation = Mathf.Clamp(xRotation, -80f, 80f); playerCamera.transform.localRotation = Quaternion.Euler(xRotation, 0f, 0f); transform.Rotate(Vector3.up * mouseX); } void HandleMovement() { float x = (MobileInputManager.Instance != null ? MobileInputManager.Instance.moveInput.x : 0) + Input.GetAxis("Horizontal"); float z = (MobileInputManager.Instance != null ? MobileInputManager.Instance.moveInput.y : 0) + Input.GetAxis("Vertical"); Vector3 move = transform.right * x + transform.forward * z; controller.Move(move * speed * Time.deltaTime); if (controller.isGrounded && velocity.y < 0) { velocity.y = -2f; } velocity.y += gravity * Time.deltaTime; controller.Move(velocity * Time.deltaTime); } void HandleShooting() { bool shootPressed = (MobileInputManager.Instance != null && MobileInputManager.Instance.isFiring) || Input.GetButton("Fire1"); if (shootPressed && Time.time >= nextTimeToFire) { nextTimeToFire = Time.time + fireRate; Shoot(); } } void Shoot() { RaycastHit hit; if (Physics.Raycast(playerCamera.transform.position, playerCamera.transform.forward, out hit, range)) { Target target = hit.transform.GetComponent(); if (target != null) { target.TakeDamage(damage); } } } } // ========================================== // 3. TARGET & HEALTH SYSTEM // Tracks object or player health and destruction on death. // ========================================== public class Target : MonoBehaviour { public float health = 100f; public void TakeDamage(float amount) { health -= amount; if (health <= 0f) { Die(); } } void Die() { Destroy(gameObject); } } // ========================================== // 4. PLAYER INVENTORY & LOOT PICKUPS // Handles item triggers on the ground and player storage. // ========================================== public class ItemPickup : MonoBehaviour { public enum ItemType { Ammo, HealthKit, Armor } public ItemType itemType; public int amount = 1; private void OnTriggerEnter(Collider other) { if (other.CompareTag("Player")) { PlayerInventory inventory = other.GetComponent(); if (inventory != null) { inventory.AddItem(itemType, amount); Destroy(gameObject); } } } } public class PlayerInventory : MonoBehaviour { public int ammoCount = 0; public int healthKits = 0; public int armorCount = 0; public void AddItem(ItemPickup.ItemType type, int amount) { switch (type) { case ItemPickup.ItemType.Ammo: ammoCount += amount; break; case ItemPickup.ItemType.HealthKit: healthKits += amount; break; case ItemPickup.ItemType.Armor: armorCount += amount; break; } } } // ========================================== // 5. SHRINKING PLAY ZONE MECHANIC // Continuously scales down the safe area zone and damages outside players. // ========================================== public class ZoneShrink : MonoBehaviour { public Vector3 targetScale = new Vector3(10f, 100f, 10f); public float shrinkDuration = 120f; public float damagePerSecond = 5f; private Vector3 initialScale; private float timer = 0f; void Start() { initialScale = transform.localScale; } void Update() { if (timer < shrinkDuration) { timer += Time.deltaTime; transform.localScale = Vector3.Lerp(initialScale, targetScale, timer / shrinkDuration); } } private void OnTriggerExit(Collider other) { if (other.CompareTag("Player")) { Target playerHealth = other.GetComponent(); if (playerHealth != null) { playerHealth.TakeDamage(damagePerSecond * Time.deltaTime); } } } }
